import { v } from "convex/values";
import { TeamsClient } from "@clipin/convex-teams";
import { components } from "./_generated/api";
import { internalMutation, query, mutation } from "./_generated/server";
import { assertProjectAccessForTeam } from "./lib/auth";
const teams = new TeamsClient(components.teams);
export const setup = internalMutation({
  args: {},
  returns: v.object({ projectId: v.id("projects"), teamSlug: v.string() }),
  handler: async ctx => {
    const teamSlug = `rehearsal-${crypto.randomUUID()}`;
    const teamPublicId = crypto.randomUUID();
    const now = Date.now();
    const sourceId = await ctx.db.insert("teams", { teamName: "Rehearsal", teamSlug, teamPublicId, ownerUserId: "rehearsal-owner", status: "active", createdAt: now, updatedAt: now });
    await ctx.db.insert("teamMemberships", { teamId: sourceId, userId: "rehearsal-owner", role: "owner", createdAt: now, updatedAt: now });
    const projectId = await ctx.db.insert("projects", { teamId: sourceId, name: "Rehearsal", slug: teamSlug, createdBy: "rehearsal-owner", createdAt: now, updatedAt: now });
    const args = { teamPublicId, teamName: "Rehearsal", teamSlug, ownerUserId: "rehearsal-owner", personal: false, expectedMemberCount: 1 };
    const first = await teams.importTeam(ctx, args);
    await teams.finishImport(ctx, teamPublicId);
    const second = await teams.importTeam(ctx, args);
    if (first.teamId !== second.teamId || second.status !== "complete") throw new Error("Import is not repeatable.");
    return { projectId, teamSlug };
  },
});
export const compare = query({
  args: { projectId: v.id("projects"), teamSlug: v.string() },
  returns: v.boolean(),
  handler: async (ctx, args) => {
    const identity = await ctx.auth.getUserIdentity();
    if (!identity) throw new Error("Unauthenticated rehearsal.");
    const source = await assertProjectAccessForTeam(ctx, args.projectId, "read");
    const imported = await teams.getTeamBySlug(ctx, identity.subject, args.teamSlug);
    if (!imported || imported.role !== source.teamRole) throw new Error("Authorization changed.");
    return true;
  },
});
export const select = mutation({
  args: { teamSlug: v.string() },
  handler: async (ctx, args) => {
    const identity = await ctx.auth.getUserIdentity();
    if (!identity) throw new Error("Not authenticated");
    await teams.setActiveTeam(ctx, identity.subject, args.teamSlug);
    const active = await teams.getActiveTeam(ctx, identity.subject);
    if (active?.slug !== args.teamSlug) throw new Error("Selection failed");
    return active;
  },
});
export const issue = mutation({
  args: { teamSlug: v.string(), email: v.string() },
  handler: async (ctx, args) => {
    const identity = await ctx.auth.getUserIdentity();
    if (!identity) throw new Error("Not authenticated");
    return await teams.createInvite(ctx, { userId: identity.subject, ...args, role: "member" });
  },
});
export const accept = mutation({
  args: { token: v.string() },
  handler: async (ctx, args) => {
    const identity = await ctx.auth.getUserIdentity();
    if (!identity?.email || identity.emailVerified !== true) throw new Error("Verified email required");
    return await teams.acceptInvite(ctx, { userId: identity.subject, email: identity.email, token: args.token, seatLimit: 2 });
  },
});
export const remove = mutation({
  args: { teamSlug: v.string(), targetUserId: v.string() },
  handler: async (ctx, args) => {
    const identity = await ctx.auth.getUserIdentity();
    if (!identity) throw new Error("Not authenticated");
    return await teams.removeMember(ctx, { userId: identity.subject, ...args });
  },
});
export const transfer = mutation({
  args: { teamSlug: v.string(), targetUserId: v.string() },
  handler: async (ctx, args) => {
    const identity = await ctx.auth.getUserIdentity();
    if (!identity) throw new Error("Not authenticated");
    return await teams.transferOwnership(ctx, { userId: identity.subject, ...args });
  },
});
export const removeTeam = mutation({
  args: { teamPublicId: v.string() },
  handler: async (ctx, args) => {
    const identity = await ctx.auth.getUserIdentity();
    if (!identity) throw new Error("Not authenticated");
    return await teams.deleteTeam(ctx, identity.subject, args.teamPublicId);
  },
});
